   ```python
   # Compute profit factor from predictions vs actual labels
   # BUY predictions (pred=1): profit if label=1 (correct), loss if label=0 (wrong)
   # SELL predictions (pred=0): profit if label=0 (correct), loss if label=1 (wrong)
   # Use simple count-based PF: gross_profit / gross_loss
   gross_profit = 0.0
   gross_loss = 0.0
   _pf_threshold = 0.55  # threshold for BUY signal
   for prob, label in zip(probas_all, test_labels, strict=True):
       if prob >= _pf_threshold:  # model predicts BUY
           if label == 1:
               gross_profit += 1.0
           else:
               gross_loss += 1.0
       # SELL predictions are not counted (strategies are long-only for PF)
   profit_factor = gross_profit / gross_loss if gross_loss > 0 else (2.0 if gross_profit > 0 else 1.0)
   ```
