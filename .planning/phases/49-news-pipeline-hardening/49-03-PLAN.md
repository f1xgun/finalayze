    1. Change `_process_news_article` return type from `None` to `bool`:
       ```python
       def _process_news_article(self, article: NewsArticle) -> bool:
           """Analyze a single article and update sentiment cache.

           Returns:
               True if LLM produced a real result, False if fallback was used.
           """
           sentiment, event = self._run_async(self._analyze_article(article))
           # Validate LLM-extracted tickers (from Task 1)
           if sentiment.tickers:
               market_id = next(iter(self._fetchers), "moex")
               validated = validate_tickers(sentiment.tickers, self._registry, market_id)

           # ... existing sentiment cache update logic unchanged ...

       ```

    2. Update `_news_cycle()` liveness tracking to use the return value:
       ```python
       ok_count = 0
       fail_count = 0
       llm_ok_count = 0  # NEW: tracks real LLM successes (not fallbacks)
       for article in articles:
           try:
               llm_succeeded = self._process_news_article(article)
               ok_count += 1
               if llm_succeeded:
                   llm_ok_count += 1
           except Exception:
               fail_count += 1
               _log.exception("_news_cycle: error processing article %s", article.id)

       # LLM liveness tracking: use llm_ok_count instead of ok_count
       if articles:
           if llm_ok_count == 0 and (ok_count + fail_count) > 0:
               # All articles either crashed or used LLM fallback
               self._llm_consecutive_failures += 1
               MetricsCollector.inc_llm_liveness_failure()

                   self._alerter.on_error(
                       "LLMLiveness",
                       f"LLM failed {self._llm_consecutive_failures} consecutive news cycles",
                   )
           else:
               self._llm_consecutive_failures = 0
       ```
