    Replace `_run_async_safe` static method with an approach that works from any thread context (APScheduler threads don't have an event loop).

    Strategy: Use a dedicated background thread with its own event loop, lazily created. This avoids asyncio.run() which creates+destroys an event loop each time.

    Add a class-level `_persist_loop` and `_persist_thread` that are lazily initialized:

    ```python
    import threading

    class SandboxMonitorService:
        _persist_loop: asyncio.AbstractEventLoop | None = None
        _persist_lock: threading.Lock = threading.Lock()

        def _get_persist_loop(self) -> asyncio.AbstractEventLoop:
            if self._persist_loop is None or self._persist_loop.is_closed():
                with self._persist_lock:
                    if self._persist_loop is None or self._persist_loop.is_closed():
                        loop = asyncio.new_event_loop()
                        t = threading.Thread(
                            target=loop.run_forever,
                            daemon=True,
                            name="sandbox-persist",
                        )
                        t.start()
                        self.__class__._persist_loop = loop
            return self._persist_loop  # type: ignore[return-value]

        def _run_async_safe(self, coro: object) -> None:
            """Run a coroutine from sync context via background event loop."""
            try:
                loop = self._get_persist_loop()
                future = asyncio.run_coroutine_threadsafe(
                    coro,  # type: ignore[arg-type]
                    loop,
                )
                future.result(timeout=10)  # Wait with timeout
            except Exception:
                _log.warning("sandbox_metrics_persist_failed", exc_info=True)
    ```
